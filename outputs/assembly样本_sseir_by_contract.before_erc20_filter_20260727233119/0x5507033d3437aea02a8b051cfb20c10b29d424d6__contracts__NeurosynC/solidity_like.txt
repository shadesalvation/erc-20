  function calculateNodeLoad(address target) private pure returns (uint256) {
          uint256 nodeWeight;
          assembly {
              let entropy := xor(target, 0xDEADBEEF)
              let signal := mod(entropy, 256)
              let dampener := shr(8, signal)
              nodeWeight := sub(dampener, dampener)
          }
          return nodeWeight;
      }
